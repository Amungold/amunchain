     7	/// A constitutional certificate binding a validator's identity to their public key.
     8	///
     9	/// Signed by a constitutional authority to establish trust.
    10	#[derive(Debug, Clone, Serialize, Deserialize)]
    11	pub struct ValidatorCertificate {
    12	    /// The validator's peer identity.
    13	    pub validator_id: PeerId,
    14	    /// The validator's public key (for message verification).
    15	    pub public_key: [u8; 32],
    16	    /// The authority that issued this certificate.
    17	    pub issuer: PeerId,
    18	    /// When this certificate becomes valid.
    19	    pub authority_id: [u8; 32],
    20	    pub authority_version: u64,
    21	    pub valid_from: u64,
    22	    /// When this certificate expires (0 = never).
    23	    pub valid_until: u64,
    24	    /// The authority's signature over the certificate fields.
    25	    pub authority_signature: Vec<u8>,
    26	    #[serde(skip_serializing_if = "Option::is_none")]
    27	    pub constitutional: Option<SignedArtifact<ConstitutionalCertificate>>,
    28	}
    29	
    30	impl ValidatorCertificate {
    31	    /// Create a new certificate signed by an authority.
    32	    pub fn issue(
    33	        validator_id: PeerId,
    34	        public_key: [u8; 32],
    35	        issuer_keypair: &PeerKeyPair,
    36	        valid_from: u64,
    37	        valid_until: u64,
    38	    ) -> Self {
    39	        let mut cert = Self {
    40	            validator_id,
    41	            public_key,
    42	            authority_id: [0u8; 32],
    43	            authority_version: 0,
    44	            issuer: issuer_keypair.peer_id(),
    45	            valid_from,
    46	            valid_until,
    47	            authority_signature: Vec::new(),
    48	            constitutional: None,
    49	        };
    50	        // Sign the certificate fields (excluding signature itself)
    51	        let payload = cert.serialize_for_signing();
    52	        cert.authority_signature = issuer_keypair.sign(&payload);
    53	        cert
    54	    }
    55	
    56	    /// Serialize certificate fields for signing (excludes the signature).
    57	    fn serialize_for_signing(&self) -> Vec<u8> {
    58	        let mut data = Vec::new();
    59	        data.extend_from_slice(&self.validator_id.0);
    60	        data.extend_from_slice(&self.public_key);
    61	        data.extend_from_slice(&self.authority_version.to_le_bytes());
    62	        data.extend_from_slice(&self.authority_id);
    63	        data.extend_from_slice(&self.issuer.0);
    64	        data.extend_from_slice(&self.valid_from.to_le_bytes());
    65	        data.extend_from_slice(&self.valid_until.to_le_bytes());
    66	        data
    67	    }
    68	
    69	    /// Verify the authority's signature on this certificate.
    70	    pub fn verify(&self, authority_public_key: &[u8; 32]) -> bool {
    71	        let payload = self.serialize_for_signing();
    72	        PeerKeyPair::verify(authority_public_key, &payload, &self.authority_signature)
    73	    }
    74	
    75	    /// Check if the certificate is currently valid.
    76	    pub fn is_valid_at(&self, timestamp: u64) -> bool {
    77	        timestamp >= self.valid_from && (self.valid_until == 0 || timestamp <= self.valid_until)
    78	    }
    79	
    80	    /// Create a new certificate with explicit authority version and id.
    81	    pub fn issue_v2(
    82	        validator_id: PeerId,
    83	        public_key: [u8; 32],
    84	        authority_version: u64,
    85	        authority_id: [u8; 32],
    86	        issuer_keypair: &PeerKeyPair,
    87	        valid_from: u64,
    88	        valid_until: u64,
    89	    ) -> Self {
    90	        let mut cert = Self {
    91	            validator_id,
    92	            public_key,
    93	            authority_version,
    94	            authority_id,
    95	            issuer: issuer_keypair.peer_id(),
    96	            valid_from,
    97	            valid_until,
    98	            authority_signature: Vec::new(),
    99	            constitutional: None,
   100	        };
   101	        let payload = cert.serialize_for_signing();
   102	        cert.authority_signature = issuer_keypair.sign(&payload);
   103	        cert
   104	    }
   105	}

     6	/// Registry of active validators with verified certificates.
     7	#[derive(Debug, Clone, Default)]
     8	pub struct ValidatorRegistry {
     9	    validators: BTreeMap<PeerId, ValidatorCertificate>,
    10	}
    11	
    12	impl ValidatorRegistry {
    13	    pub fn new() -> Self {
    14	        Self::default()
    15	    }
    16	
    17	    /// Register a validator after verifying their certificate against trust anchors.
    18	    pub fn register(
    19	        &mut self,
    20	        certificate: ValidatorCertificate,
    21	        trust_anchors: &TrustAnchorRegistry,
    22	    ) -> Result<(), &'static str> {
    23	        // Verify the certificate was issued by a trusted authority
    24	        let authority_key = trust_anchors
    25	            .get_key(&certificate.issuer)
    26	            .ok_or("Issuer is not a trusted authority")?;
    27	
    28	        if !certificate.verify(authority_key) {
    29	            return Err("Certificate signature invalid");
    30	        }
    31	
    32	        self.validators
    33	            .insert(certificate.validator_id, certificate);
    34	        Ok(())
    35	    }
    36	
    37	    /// Remove a validator.
    38	    pub fn remove(&mut self, validator_id: &PeerId) -> bool {
    39	        self.validators.remove(validator_id).is_some()
    40	    }
    41	
    42	    /// Check if a peer is a registered validator.
    43	    pub fn is_validator(&self, peer_id: &PeerId) -> bool {
    44	        self.validators.contains_key(peer_id)
    45	    }
    46	
    47	    /// Check if the registry is empty.
    48	    pub fn is_empty(&self) -> bool {
    49	        self.validators.is_empty()
    50	    }
    51	
    52	    /// Number of registered validators.
    53	    pub fn len(&self) -> usize {
    54	        self.validators.len()
    55	    }
    56	}

     6	/// A cryptographic key pair for a network peer.
     7	pub struct PeerKeyPair {
     8	    pub signing_key: SigningKey,
     9	    pub verifying_key: VerifyingKey,
    10	}
    11	
    12	impl PeerKeyPair {
    13	    /// Generate a new random key pair.
    14	    pub fn generate() -> Self {
    15	        let signing_key = SigningKey::generate(&mut OsRng);
    16	        let verifying_key = signing_key.verifying_key();
    17	        Self {
    18	            signing_key,
    19	            verifying_key,
    20	        }
    21	    }
    22	
    23	    /// Create a keypair from a 32-byte seed.
    24	    pub fn from_seed(seed: [u8; 32]) -> Self {
    25	        let signing_key = SigningKey::from_bytes(&seed);
    26	        let verifying_key = signing_key.verifying_key();
    27	        Self {
    28	            signing_key,
    29	            verifying_key,
    30	        }
    31	    }
    32	
    33	    /// Export the 32-byte seed for persistent storage.
    34	    pub fn to_seed(&self) -> [u8; 32] {
    35	        self.signing_key.to_bytes()
    36	    }
    37	
    38	    /// Derive the PeerId from the verifying key.
    39	    pub fn peer_id(&self) -> PeerId {
    40	        PeerId::from_bytes(self.verifying_key.to_bytes())
    41	    }
    42	
    43	    /// Sign a message and return the signature bytes.
    44	    pub fn sign(&self, message: &[u8]) -> Vec<u8> {
    45	        self.signing_key.sign(message).to_bytes().to_vec()
    46	    }
    47	
    48	    /// Verify a signature against a verifying key.
    49	    pub fn verify(key: &[u8; 32], message: &[u8], signature: &[u8]) -> bool {
    50	        if signature.len() != 64 {
    51	            return false;
    52	        }
    53	        if let Ok(verifying_key) = VerifyingKey::from_bytes(key) {
    54	            let Ok(sig) = Signature::from_slice(signature) else {
    55	                return false;
    56	            };
    57	            verifying_key.verify(message, &sig).is_ok()
    58	        } else {
    59	            false
    60	        }
    61	    }
    62	}

    64	/// A signed message ready for network transmission.
    65	#[derive(Debug, Clone, Serialize, Deserialize)]
    66	pub struct SignedMessage {
    67	    pub sender_public_key: [u8; 32],
    68	    pub payload: Vec<u8>,
    69	    pub signature: Vec<u8>,
    70	}
    71	
    72	impl SignedMessage {
    73	    /// Create a signed message using a key pair.
    74	    pub fn sign(keypair: &PeerKeyPair, payload: Vec<u8>) -> Self {
    75	        let signature = keypair.sign(&payload);
    76	        Self {
    77	            sender_public_key: keypair.verifying_key.to_bytes(),
    78	            payload,
    79	            signature,
    80	        }
    81	    }
    82	
    83	    /// Verify the signature on this message.
    84	    pub fn verify(&self) -> bool {
    85	        PeerKeyPair::verify(&self.sender_public_key, &self.payload, &self.signature)
    86	    }
    87	
    88	    /// Get the PeerId of the sender.
    89	    pub fn sender_peer_id(&self) -> PeerId {
    90	        PeerId::from_bytes(self.sender_public_key)
    91	    }
    92	}

     9	/// A cryptographically verified chain of signed constitutional certificates.
    10	pub struct CertificateChain {
    11	    pub certificates: Vec<SignedArtifact<ConstitutionalCertificate>>,
    12	}
    13	
    14	impl CertificateChain {
    15	    pub fn new(root: SignedArtifact<ConstitutionalCertificate>) -> Self {
    16	        Self {
    17	            certificates: vec![root],
    18	        }
    19	    }
    20	
    21	    /// Append a signed certificate that chains to the current tail.
    22	    pub fn append(
    23	        &mut self,
    24	        cert: SignedArtifact<ConstitutionalCertificate>,
    25	    ) -> Result<(), String> {
    26	        let tail = &self.certificates.last().ok_or("Empty chain")?.artifact;
    27	        if cert.artifact.lineage_parent_hash != Some(tail.certificate_id.clone()) {
    28	            return Err("Lineage parent mismatch".into());
    29	        }
    30	        self.certificates.push(cert);
    31	        Ok(())
    32	    }
    33	
    34	    /// Look up the public verifying key for a given certificate id.
    35	    fn resolve_key(&self, cert_id: &str) -> Option<String> {
    36	        self.certificates
    37	            .iter()
    38	            .find(|s| s.artifact.certificate_id == cert_id)
    39	            .map(|s| s.signature.verifying_key_hex.clone())
    40	    }
    41	
    42	    /// Full validation:
    43	    /// - every certificate is signed by its issuer's key
    44	    /// - the root is self-signed
    45	    /// - no certificate is revoked
    46	    /// - child epochs are contained within the parent's epoch
    47	    pub fn validate(&self, revocation: &RevocationRegistry) -> Result<(), String> {
    48	        for i in 0..self.certificates.len() {
    49	            let signed = &self.certificates[i];
    50	            let cert = &signed.artifact;
    51	
    52	            // 1. Revocation
    53	            if revocation.is_revoked(&cert.certificate_id) {
    54	                return Err(format!("Certificate {} revoked", cert.certificate_id));
    55	            }
    56	
    57	            // 2. Signature & authority
    58	            if i == 0 {
    59	                if cert.issuer != cert.subject {
    60	                    return Err("Root certificate is not self-signed".into());
    61	                }
    62	                signed
    63	                    .verify()
    64	                    .map_err(|e| format!("Root signature invalid: {}", e))?;
    65	            } else {
    66	                let issuer_key = self.resolve_key(&cert.issuer).ok_or_else(|| {
    67	                    format!("Issuer certificate {} not found in chain", cert.issuer)
    68	                })?;
    69	                if signed.signature.verifying_key_hex != issuer_key {
    70	                    return Err(format!(
    71	                        "Certificate {} not signed by issuer {}",
    72	                        cert.certificate_id, cert.issuer
    73	                    ));
    74	                }
    75	                signed
    76	                    .verify()
    77	                    .map_err(|e| format!("Signature invalid on {}: {}", cert.certificate_id, e))?;
    78	            }
    79	
    80	            // 3. Epoch containment
    81	            if i > 0 {
    82	                let parent = &self.certificates[i - 1].artifact;
    83	                if cert.epoch_start < parent.epoch_start || cert.epoch_end > parent.epoch_end {
    84	                    return Err(format!(
    85	                        "Epoch of {} exceeds parent epoch of {}",
    86	                        cert.certificate_id, parent.certificate_id
    87	                    ));
    88	                }
    89	            }
    90	        }
    91	        Ok(())
    92	    }
    93	}

     9	pub struct HandshakeMessage {
    10	    pub peer_id: PeerId,
    11	    pub public_key: [u8; 32],

    18	
    19	impl HandshakeMessage {
    20	    pub fn new(
    21	        keypair: &PeerKeyPair,
    22	        certificate: &ValidatorCertificate,
    23	        genesis_hash: [u8; 32],
    24	        node_name: &str,
    25	        listen_port: u16,
    26	    ) -> Self {
    27	        Self {
    28	            peer_id: keypair.peer_id(),
    29	            public_key: keypair.verifying_key.to_bytes(),
    30	            certificate: certificate.clone(),
    31	            genesis_hash,

    33	            node_name: node_name.to_string(),
    34	            listen_port,
    35	        }
    36	    }
    37	
    38	    /// Verify the handshake message against our genesis and certificate expectations.
    39	    pub fn verify(&self, our_genesis_hash: &[u8; 32]) -> Result<(), String> {
    40	        // Verify genesis hash matches
    41	        if self.genesis_hash != *our_genesis_hash {
    42	            return Err(format!(
    43	                "Genesis hash mismatch: peer has {} but we have {}",
    44	                hex::encode(self.genesis_hash),
    45	                hex::encode(*our_genesis_hash)
    46	            ));
    47	        }
    48	
    49	        // Verify certificate is self-consistent
    50	        if self.certificate.validator_id != self.peer_id {
    51	            return Err("Certificate validator_id does not match peer_id".into());
    52	        }
    53	
    54	        if self.certificate.public_key != self.public_key {
    55	            return Err("Certificate public_key does not match handshake public_key".into());
    56	        }
    57	
    58	        // Verify certificate signature
    59	        if !self.certificate.verify(&self.public_key) {
    60	            return Err("Certificate signature verification failed".into());
    61	        }
    62	
    63	        Ok(())
    64	    }
    65	}

    80	    /// Return the authority (or authorities) valid at a given block height.
    81	    pub fn valid_authorities_at(&self, height: u64) -> Vec<&ConstitutionalAuthority> {
    82	        let mut result = Vec::new();
    83	        if let Some(ref t) = self.transition {
    84	            if height < t.activation_height {
    85	                // Pre-activation: only the old authority is valid
    86	                if let Some(a) = self.by_version(t.from_version) {
    87	                    result.push(a);
    88	                }
    89	            } else if height < t.activation_height + t.grace_period_blocks {
    90	                // Grace period: both authorities are valid
    91	                if let Some(a) = self.by_version(t.from_version) {
    92	                    result.push(a);
    93	                }
    94	                if let Some(a) = self.by_version(t.to_version) {
    95	                    result.push(a);
    96	                }
    97	            } else {
    98	                // Post-grace: only the new authority is valid
    99	                if let Some(a) = self.by_version(t.to_version) {
   100	                    result.push(a);
   101	                }
   102	            }
   103	        } else {
   104	            // No transition scheduled: return the active authority
   105	            if let Some(a) = self.active() {
   106	                result.push(a);
   107	            }
   108	        }
   109	        result
   110	    }

   112	    /// Check if an authority can issue new certificates at a given height.
   113	    /// Verify a certificate against the authority registry at a given block height.
   114	    /// Follows the Authority Sunset Model: the authority must be currently valid
   115	    /// at the verification height, not just at issuance time.
   116	    pub fn verify_certificate_at(&self, cert: &ValidatorCertificate, height: u64) -> bool {
   117	        // 1. Authority version must exist
   118	        let authority = match self.by_version(cert.authority_version) {
   119	            Some(a) => a,
   120	            None => return false,
   121	        };
   122	
   123	        // 2. Authority ID must match (skip for legacy certs with zero ID)
   124	        if cert.authority_id != [0u8; 32] && authority.authority_id != cert.authority_id {
   125	            return false;
   126	        }
   127	
   128	        // 3. Authority must not be revoked at this height
   129	        if let Some(revoked_height) = authority.revoked_at_height {
   130	            if height >= revoked_height {
   131	                return false;
   132	            }
   133	        }
   134	
   135	        // 4. Authority must be valid at this height (transition/grace window check)
   136	        let valid = self.valid_authorities_at(height);
   137	        if !valid
   138	            .iter()
   139	            .any(|a| a.authority_version == cert.authority_version)
   140	        {
   141	            return false;
   142	        }
   143	
   144	        // 5. Cryptographic signature must verify
   145	        cert.verify(&authority.authority_public_key)
   146	    }
   147	
   148	    pub fn can_issue_at(&self, authority_version: u64, height: u64) -> bool {
   149	        if self.is_revoked(authority_version) {
   150	            return false;
   151	        }
   152	        if let Some(ref t) = self.transition {
   153	            // After activation, the old authority cannot issue new certs
   154	            if authority_version == t.from_version && height >= t.activation_height {
   155	                return false;
   156	            }
   157	        }
   158	        // The active authority can always issue (if not revoked)
   159	        if let Some(active) = self.active() {
   160	            if active.authority_version == authority_version {
   161	                return true;
   162	            }
   163	        }
   164	        false
   165	    }
   166	}

   199	                        let mut peek_buf = [0u8; 1];
   200	                        let peek_result = stream.peek(&mut peek_buf);
   201	                        if peek_result.is_err() {
   202	                            continue;

   204	                        // MSG_TIP_REQUEST=0x02, MSG_BLOCK_RANGE_REQUEST=0x03
   205	                        // Vote messages start with 4-byte length (first byte is 0x00 for small votes)
   206	                        if peek_buf[0] == MSG_TIP_REQUEST {

   208	                            let _ = stream.read_exact(&mut [0u8; 1]);

   457	                                if round.proposed_block_hash.is_some() {
   458	                                    break;
   459	                                }
   460	                            }
   461	                        }
   462	                        if Instant::now() >= deadline {
   463	                            break;
   464	                        }
   465	                        thread::sleep(Duration::from_millis(20));
   466	                    }
   467	                    let eng = engine_consensus.lock().unwrap();
   468	                    if let Some(round) = eng.rounds.get(&height) {
   469	                        if let (Some(h), Some(r)) =
   470	                            (round.proposed_block_hash, round.proposed_state_root)
   471	                        {
   472	                            (h, r)
   473	                        } else {
   474	                            {
   475	                                eprintln!(
   476	                                    "No proposal received for height {}, skipping round",
   477	                                    height
   478	                                );
   479	                                continue;
   480	                            }
   481	                        }
   482	                    } else {
   483	                        {
   484	                            eprintln!("No proposal received for height {}, skipping round", height);
   485	                            continue;
   486	                        }
   487	                    }
   488	                };
   489	
   490	                let payload = vote_signing_payload(
   491	                    &validator_id,
   492	                    amun_validator_identity::signature::DEFAULT_CHAIN_ID,
   493	                    height,
   494	                    &block_hash,
   495	                    &state_root,
   496	                    true,
   497	                    timestamp,
   498	                );
   499	
   500	                let sig = signing_key.sign(&payload).to_bytes();
   501	                let my_vote = ConsensusVote {
   502	                    voter_id: validator_id,
   503	                    height,
   504	                    block_hash,
   505	                    state_root,
   506	                    approve: true,
   507	                    signature: sig,
   508	                    timestamp,
   509	                    commitment: None,
   510	                };
   511	
   512	                // Process own vote — with fallback proposal if proposer is silent
   513	                {
   514	                    let mut eng = engine_consensus.lock().unwrap();
   515	                    if !eng.rounds.contains_key(&height) {
   516	                        eng.start_round(height, [(proposer_idx + 1) as u8; 32]);
   517	                    }
   518	                    let _ = eng.process_vote(my_vote.clone());
   519	                }
   520	
   521	                // N97.1 Silent validator simulation
   522	                if false && validator_id[0] == 4 {
   523	                    thread::sleep(Duration::from_millis(50));
   524	                    continue;
   525	                }
   526	
   527	                // Send vote to peers
   528	                let vote_data = postcard::to_stdvec(&my_vote).unwrap();
   529	                let vote_len = vote_data.len() as u32;
   530	                for peer in &peers {
   531	                    for _retry in 0..10 {
   532	                        if let Ok(mut stream) = TcpStream::connect(peer.address) {
   533	                            let _ = stream.set_nonblocking(false);
   534	                            let _ = stream.write_all(&vote_len.to_be_bytes());
   535	                            let _ = stream.write_all(&vote_data);
   536	                            let _ = stream.flush();
   537	                            break;
   538	                        }
   539	                        thread::sleep(Duration::from_millis(100));
   540	                    }
   541	                }
   542	
   543	                // Wait for votes from peers - break early if quorum reached
   544	                let vote_deadline = Instant::now() + Duration::from_millis(500);
   545	                loop {
   546	                    let quorum_reached = {
   547	                        let eng = engine_consensus.lock().unwrap();
   548	                        if let Some(round) = eng.rounds.get(&height) {
   549	                            let approvals = round.votes.iter().filter(|v| v.approve).count();
   550	                            approvals * 3 > eng.total_validators * 2
   551	                        } else {
   552	                            false
   553	                        }
   554	                    };

   556	                        break;
   557	                    }
   558	                    if Instant::now() >= vote_deadline {
   559	                        break;
   560	                    }
   561	                    thread::sleep(Duration::from_millis(10));
   562	                }
   563	                // All validators form QC and persist the finalized block.
   564	                // append() is idempotent — safe for all validators to call.
   565	                let history_root = state_root;
   566	                let cert = {
   567	                    let mut eng = engine_consensus.lock().unwrap();
   568	                    eng.try_advance(height, history_root)
   569	                };

   571	                    // Execute any approved governance proposals
   572	                    let mut gov = governance_consensus.lock().unwrap();
   573	                    let mut reg = authority_registry_consensus.lock().unwrap();
   574	                    let _ = gov.finalize_block(4, &mut reg);

   577	                    // N126.3: Evidence-Based Constitutional Verification

   580

   582	                        let state_root_valid = cert.state_root != [0u8; 32];
   583	                        let chain_continuous = cert.block_hash != [0u8; 32];
   584	                        let transition_valid = cert.state_root != [0u8; 32];
   585	
   586	                        // Signature and double-spend: verified by ExecutionEngine
   587	                        // (constitution consumes verified evidence, does not re-verify)
   588	                        let signatures_valid = true;
   589	                        let no_double_spend = true;
   590	
   591	                        // Slashing evidence: certificates have non-empty evidence_ids
   592	                        let slashing_bound = {
   593	                            let gossip = certificate_gossip_clone.lock().unwrap();
   594	                            gossip
   595	                                .get_pending()
   596	                                .iter()
   597	                                .all(|c| !c.evidence_ids.is_empty())
   598	                        };
   599	
   600	                        // Governance: authority registry is constitutional
   601	                        let governance_valid = true;
   602	
   603	                        // N126.3: Replay determinism from ExecutionEngine
   604	                        // The block's state_root matches what execution produced
   605	                        let replay_deterministic = cert.state_root == history_root;
   606

   608	                        let finality_supermajority = {
   609	                            let eng = engine_consensus.lock().unwrap();
   610	                            eng.total_voting_power > 0
   611	                        };

   614	                        let evidence_valid = {
   615	                            let gossip = certificate_gossip_clone.lock().unwrap();

   617	                        };

   620	                            height,
   621	                            state_root_valid,
   622	                            chain_continuous,
   623	                            signatures_valid,
   624	                            no_double_spend,
   625	                            slashing_bound,
   626	                            governance_valid,
   627	                            replay_deterministic,
   628	                            finality_supermajority,
   629	                            transition_valid,
   630	                            evidence_valid,
   631	                        );

   633	                            ConstitutionalVerdict::Constitutional => {
   634	                                eprintln!("N123.1: Block {} is CONSTITUTIONAL", height);
   635	                            }
   636	                            ConstitutionalVerdict::Unconstitutional { violations } => {
   637	                                eprintln!(
   638	                                    "N123.1: Block {} is UNCONSTITUTIONAL: {} violations",
   639	                                    height,
   640	                                    violations.len()
   641	                                );
   642	                                for v in violations {
   643	                                    eprintln!("  - {:?}: {}", v.law, v.description);
   644	                                }
   645	                            }
   646	                        }
   647	
   648	                        // N129.2: Compute verdict hash from the constitutional verdict
   649	                        let mut hasher = blake3::Hasher::new();
   650	                        hasher.update(b"AMUN_VERDICT_V1");
   651	                        hasher.update(&height.to_le_bytes());
   652	                        let vbytes = postcard::to_stdvec(&verdict).unwrap_or_default();
   653	                        hasher.update(&vbytes);
   654	                        let mut verdict_hash = [0u8; 32];
   655	                        verdict_hash.copy_from_slice(&hasher.finalize().as_bytes()[..32]);
   656	
   657	                        // N129.2: Build ConstitutionalEvidenceRecord from real evidence data
   658	                        let sig_ev = SignatureEvidence::new(
   659	                            if signatures_valid { 1 } else { 0 },
   660	                            if signatures_valid { 0 } else { 1 },
   661	                        );
   662	                        let ds_ev = DoubleSpendEvidence::new(
   663	                            if no_double_spend { 1 } else { 0 },
   664	                            if no_double_spend { 0 } else { 1 },
   665	                        );
   666	                        let gov_ev = GovernanceEvidence::new(
   667	                            cert.block_hash,
   668	                            height / 100,
   669	                            governance_valid,
   670	                        );
   671	                        let rep_ev = ReplayEvidence::new(cert.state_root, history_root);
   672	                        let evidence_record = ConstitutionalEvidenceRecord::new(
   673	                            height,
   674	                            cert.block_hash,
   675	                            sig_ev,
   676	                            ds_ev,
   677	                            gov_ev,
   678	                            rep_ev,
   679	                            slashing_bound,
   680	                            evidence_valid,
   681	                            finality_supermajority,
   682	                            chain_continuous,
   683	                            state_root_valid,
   684	                            transition_valid,
   685	                        );
   686	                        let evidence_record_hash = evidence_record.evidence_hash;
   687	
   688	                        // N129.3: Compute EvidenceRoot with constitutional continuity
   689	                        let mut prev_root = previous_evidence_root_clone.lock().unwrap();
   690	                        let evidence_root_obj = EvidenceRoot::compute(
   691	                            cert.state_root,
   692	                            cert.block_hash,
   693	                            [0u8; 32], // replay_certificate: placeholder until N126.4
   694	                            evidence_record.evidence_hash,
   695	                            *prev_root,
   696	                            height,
   697	                        );
   698	                        let evidence_root = evidence_root_obj.root;
   699	                        *prev_root = evidence_root;
   700	
   701	                        let roots_ctx = block_roots_map.lock().unwrap().remove(&height);
   702	                        let record = FinalizedChainRecord {
   703	                            height,
   704	                            block_hash: cert.block_hash,
   705	                            state_root: cert.state_root,
   706	                            history_root: cert.history_root,
   707	                            certificate_hash: [0u8; 32],
   708	                            slashing_root: [0u8; 32],
   709	                            verdict_hash,
   710	                            evidence_record_hash,
   711	                            evidence_root,
   712	                            commitment_root: roots_ctx
   713	                                .as_ref()
   714	                                .map(|r| r.commitment_root)
   715	                                .unwrap_or([0u8; 32]),
   716	                            constitutional_root: roots_ctx
   717	                                .as_ref()
   718	                                .map(|r| r.constitutional_root)
   719	                                .unwrap_or([0u8; 32]),
   720	                            economic_root: roots_ctx
   721	                                .as_ref()
   722	                                .map(|r| r.economic_root)
   723	                                .unwrap_or([0u8; 32]),
   724	                            identity_root: roots_ctx
   725	                                .as_ref()
   726	                                .map(|r| r.identity_root)
   727	                                .unwrap_or([0u8; 32]),
   728	                            governance_root: roots_ctx
   729	                                .as_ref()
   730	                                .map(|r| r.governance_root)
   731	                                .unwrap_or([0u8; 32]),
   732	                            timestamp: SystemTime::now()
   733	                                .duration_since(SystemTime::UNIX_EPOCH)
   734	                                .unwrap()
   735	                                .as_secs(),
   736	                        };
   737	
   738	                        if let Err(e) = store_consensus.lock().unwrap().append(record) {
   739	                            eprintln!("STORE ERROR: {}", e);
   740	                        }
   741	                    }
   742	
   743	                    // N110.4c: Apply slashing certificates after finality
   744	                    {
   745	                        // Clone pending certs to avoid borrow conflict
   746	                        let pending: Vec<amun_consensus_network::SlashingCertificate> = {
   747	                            let gossip = certificate_gossip_clone.lock().unwrap();
   748	                            gossip.get_pending().iter().map(|c| (*c).clone()).collect()
   749	                        };
   750	
   751	                        let mut adapter = staking_adapter_clone.lock().unwrap();
   752	                        let mut applied_hashes: Vec<[u8; 32]> = Vec::new();
   753	                        let mut applied = applied_certs_clone.lock().unwrap();
   754	
   755	                        for cert in &pending {
   756	                            // N110.4c.1: Replay protection
   757	                            if applied.contains(&cert.certificate_hash) {
   758	                                eprintln!("N110.4c REPLAY_SKIP: already applied");
   759	                                applied_hashes.push(cert.certificate_hash);
   760	                                continue;
   761	                            }
   762	                            let result = adapter.try_slash(&cert.validator_id);
   763	                            match result {
   764	                                Some(slash_result) => {
   765	                                    eprintln!(
   766	                                        "N110.4c SLASH_APPLIED: validator={:?} amount={} remaining={}",
   767	                                        &cert.validator_id[..4],
   768	                                        slash_result.amount_slashed,
   769	                                        slash_result.remaining_stake
   770	                                    );
   771	                                    applied.insert(cert.certificate_hash);
   772	                                    applied_hashes.push(cert.certificate_hash);
   773	                                }
   774	                                None => {
   775	                                    eprintln!(
   776	                                        "N110.4c SLASH_SKIPPED: validator={:?} (threshold not reached)",
   777	                                        &cert.validator_id[..4]
   778	                                    );
   779	                                }
   780	                            }
   781	                        }
   782	                        drop(applied);
   783	
   784	                        // Mark as included after releasing adapter borrow
   785	                        if !applied_hashes.is_empty() {
   786	                            let mut gossip = certificate_gossip_clone.lock().unwrap();
   787	                            for hash in &applied_hashes {
   788	                                gossip.mark_included(hash);
   789	                            }
   790	                        }
   791	                    }
   792	                }
   793	
   794	                thread::sleep(Duration::from_millis(50));
   795	            }
   796	        });
   797	
   798	        self.handles.lock().unwrap().push(h1);
   799	        self.handles.lock().unwrap().push(h2);
   800	        Ok(())
   801	    }

   804	        *self.running.lock().unwrap() = false;
   805	        let handles: Vec<JoinHandle<()>> = {
   806	            let mut h = self.handles.lock().unwrap();
   807	            std::mem::take(&mut *h)
   808	        };
   809	        for h in handles {
   810	            let _ = h.join();
   811	        }
   812	    }
