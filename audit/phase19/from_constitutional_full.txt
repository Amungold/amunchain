   138	    /// Create a new certificate signed by an authority.
   139	    pub fn issue(
   140	        validator_id: PeerId,
   141	        public_key: [u8; 32],
   142	        issuer_keypair: &PeerKeyPair,
   143	        valid_from: u64,
   144	        valid_until: u64,
   145	    ) -> Self {
   146	        let mut cert = Self {
   147	            validator_id,
   148	            public_key,
   149	            authority_id: [0u8; 32],
   150	            authority_version: 0,
   151	            issuer: issuer_keypair.peer_id(),
   152	            valid_from,
   153	            valid_until,
   154	            authority_signature: Vec::new(),
   155	            constitutional: None,
   156	
   157	            runtime: RuntimeCertificateView {
   158	                peer_id: validator_id,
   159	                public_key,
   160	                authority_id: [0u8; 32],
   161	                authority_version: 0,
   162	                valid_from,
   163	                valid_until,
   164	                authority_signature: Vec::new(),
   165	            },
   166	        };
   167	        // Sign the certificate fields (excluding signature itself)
   168	        let payload = cert.serialize_for_signing();
   169	        cert.authority_signature = issuer_keypair.sign(&payload);
   170	        cert.runtime.authority_signature = cert.authority_signature.clone();
   171	        cert
   172	    }
   173	
   174	    /// Serialize certificate fields for signing (excludes the signature).
   175	    fn serialize_for_signing(&self) -> Vec<u8> {
   176	        let mut data = Vec::new();
   177	        data.extend_from_slice(&self.validator_id.0);
   178	        data.extend_from_slice(&self.public_key);
   179	        data.extend_from_slice(&self.authority_version.to_le_bytes());
   180	        data.extend_from_slice(&self.authority_id);
   181	        data.extend_from_slice(&self.issuer.0);
   182	        data.extend_from_slice(&self.valid_from.to_le_bytes());
   183	        data.extend_from_slice(&self.valid_until.to_le_bytes());
   184	        data
   185	    }

   187	    /// Verify the authority's signature on this certificate.
   188	    pub fn verify(&self, authority_public_key: &[u8; 32]) -> bool {
   189	        let payload = self.serialize_for_signing();
   190	        PeerKeyPair::verify(authority_public_key, &payload, &self.authority_signature)
   191	    }

   195	        timestamp >= self.valid_from && (self.valid_until == 0 || timestamp <= self.valid_until)
   196	    }
