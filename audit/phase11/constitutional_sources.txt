     5	use amun_constitution_builder::{canonical_bytes::CanonicalSerialize, digest::ArtifactDigest};
     6	use serde::{Deserialize, Serialize};
     7	
     8	/// A constitutional certificate whose identity is derived from a
     9	/// stable subset of its fields (everything except `certificate_id`,
    10	/// `issuer`, and `subject`).  This guarantees that the certificate
    11	/// can reference its own identity without creating a circular
    12	/// dependency.
    13	#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
    14	#[serde(rename_all = "snake_case")]
    15	pub struct ConstitutionalCertificate {
    16	    pub schema_version: u32,
    17	    pub certificate_id: String, // hex BLAKE3 of identity bytes
    18	    pub issuer: String,         // certificate_id of the issuer
    19	    pub subject: String,        // certificate_id of the subject
    20	    pub subject_verifying_key_hex: String,
    21	    pub lineage_parent_hash: Option<String>,
    22	    pub epoch_start: String,
    23	    pub epoch_end: String,
    24	    pub constitutional_scope: String,
    25	    pub timestamp: String,
    26	}
    27	
    28	impl ArtifactDigest for ConstitutionalCertificate {
    29	    fn domain_separator(&self) -> &'static [u8] {
    30	        b"AMUN_CERTIFICATE_V1"
    31	    }
    32	
    33	    fn constitutional_digest(&self) -> [u8; 32] {
    34	        let mut hasher = blake3::Hasher::new();
    35	        hasher.update(self.domain_separator());
    36	        hasher.update(&self.identity_bytes());
    37	        *hasher.finalize().as_bytes()
    38	    }
    39	}
    40	
    41	impl ConstitutionalCertificate {
    42	    // ---------------------------------------------------------------
    43	    // Constructors – roles are separated so invariants are clear
    44	    // ---------------------------------------------------------------
    45	
    46	    /// Create a self-signed root certificate.
    47	    pub fn new_root(
    48	        subject_public_key_hex: String,
    49	        epoch_start: String,
    50	        epoch_end: String,
    51	        scope: String,
    52	        timestamp: String,
    53	    ) -> Self {
    54	        let mut tmp = Self::empty();
    55	        tmp.subject_verifying_key_hex = subject_public_key_hex;
    56	        tmp.epoch_start = epoch_start;
    57	        tmp.epoch_end = epoch_end;
    58	        tmp.constitutional_scope = scope;
    59	        tmp.timestamp = timestamp;
    60	
    61	        let id = tmp.compute_id();
    62	        tmp.certificate_id = id.clone();
    63	        tmp.issuer = id.clone();
    64	        tmp.subject = id;
    65	        tmp
    66	    }
    67	
    68	    /// Create a child certificate issued by a parent.
    69	    pub fn new_child(
    70	        issuer_id: String, // parent certificate_id
    71	        subject_public_key_hex: String,
    72	        lineage_parent: String, // same as issuer_id
    73	        epoch_start: String,
    74	        epoch_end: String,
    75	        scope: String,
    76	        timestamp: String,
    77	    ) -> Self {
    78	        let mut tmp = Self::empty();
    79	        tmp.issuer = issuer_id;
    80	        tmp.subject_verifying_key_hex = subject_public_key_hex;
    81	        tmp.lineage_parent_hash = Some(lineage_parent);
    82	        tmp.epoch_start = epoch_start;
    83	        tmp.epoch_end = epoch_end;
    84	        tmp.constitutional_scope = scope;
    85	        tmp.timestamp = timestamp;
    86	
    87	        let id = tmp.compute_id();
    88	        tmp.certificate_id = id.clone();
    89	        tmp.subject = id;
    90	        tmp
    91	    }
    92	
    93	    // ---------------------------------------------------------------
    94	    // Identity helpers
    95	    // ---------------------------------------------------------------
    96	
    97	    /// Build a bare certificate used as a template for identity computation.
    98	    fn empty() -> Self {
    99	        Self {
   100	            schema_version: 1,
   101	            certificate_id: String::new(),
   102	            issuer: String::new(),
   103	            subject: String::new(),
   104	            subject_verifying_key_hex: String::new(),
   105	            lineage_parent_hash: None,
   106	            epoch_start: String::new(),
   107	            epoch_end: String::new(),
   108	            constitutional_scope: String::new(),
   109	            timestamp: String::new(),
   110	        }
   111	    }
   112	
   113	    /// Compute the certificate id from the canonical identity bytes.
   114	    fn compute_id(&self) -> String {
   115	        let bytes = self.identity_bytes();
   116	        let mut hasher = blake3::Hasher::new();
   117	        hasher.update(b"AMUN_CERTIFICATE_V1");
   118	        hasher.update(&bytes);
   119	        hex::encode(hasher.finalize().as_bytes())
   120	    }
   121	
   122	    /// Canonical bytes used for identity hashing.
   123	    ///
   124	    /// This deliberately omits `certificate_id`, `issuer`, and `subject`
   125	    /// because those fields depend on the identity itself for root
   126	    /// certificates, which would create a circular dependency.
   127	    fn identity_bytes(&self) -> Vec<u8> {
   128	        let mut c = self.clone();
   129	        c.certificate_id = String::new();
   130	        c.issuer = String::new();
   131	        c.subject = String::new();
   132	        serde_json::to_vec(&c).expect("Identity serialization must not fail")
   133	    }
   134	
   135	    /// Convenience: return the full artifact digest (including all fields).
   136	    pub fn full_digest_hex(&self) -> String {
   137	        self.digest_hex()
   138	    }
   139	}
   140	
   141	impl CanonicalSerialize for ConstitutionalCertificate {
   142	    fn canonical_bytes(&self) -> Vec<u8> {
   143	        serde_json::to_vec(self).expect("Canonical serialization must not fail")
   144	    }
   145	}

================ SignedArtifact ================
     1	use crate::keys::ConstitutionalKeyPair;
     2	use crate::signature::ConstitutionalSignature;
     3	use amun_constitution_builder::digest::ArtifactDigest;
     4	use ed25519_dalek::{Verifier, VerifyingKey};
     5	use serde::{Deserialize, Serialize};
     6	
     7	#[derive(Clone, Serialize, Deserialize)]
     8	pub struct SignedArtifact<T: ArtifactDigest> {
     9	    pub artifact: T,
    10	    pub signature: ConstitutionalSignature,
    11	}
    12	
    13	impl<T: ArtifactDigest> SignedArtifact<T> {
    14	    pub fn sign(artifact: T, keypair: &ConstitutionalKeyPair) -> Self {
    15	        let digest = artifact.constitutional_digest();
    16	        let sig = keypair.sign(&digest);
    17	        let verifying_hex = keypair.verifying_key_hex();
    18	        Self {
    19	            artifact,
    20	            signature: ConstitutionalSignature::new(sig, verifying_hex),
    21	        }
    22	    }
    23	
    24	    pub fn verify(&self) -> Result<(), String> {
    25	        let key_bytes = hex::decode(&self.signature.verifying_key_hex)
    26	            .map_err(|e| format!("Invalid verifying key hex: {}", e))?;
    27	        let arr: [u8; 32] = key_bytes
    28	            .try_into()
    29	            .map_err(|_| "Verifying key must be 32 bytes".to_string())?;
    30	        let verifying_key =
    31	            VerifyingKey::from_bytes(&arr).map_err(|e| format!("Invalid verifying key: {}", e))?;
    32	
    33	        let digest = self.artifact.constitutional_digest();
    34	        verifying_key
    35	            .verify(&digest, self.signature.dalek_signature())
    36	            .map_err(|e| format!("Signature verification failed: {}", e))
    37	    }
    38	}
    39	
    40	// Manual PartialEq + Debug to satisfy DelegationChain derives
    41	impl<T: ArtifactDigest + PartialEq> PartialEq for SignedArtifact<T> {
    42	    fn eq(&self, other: &Self) -> bool {
    43	        self.artifact == other.artifact
    44	            && self.signature.verifying_key_hex == other.signature.verifying_key_hex
    45	            && self.signature.to_hex() == other.signature.to_hex()
    46	    }
    47	}
    48	
    49	impl<T: ArtifactDigest + std::fmt::Debug> std::fmt::Debug for SignedArtifact<T> {
    50	    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    51	        f.debug_struct("SignedArtifact")
    52	            .field("artifact", &self.artifact)
    53	            .field("signature", &self.signature.verifying_key_hex)
    54	            .finish()
    55	    }
    56	}

================ Authority Registry ================
     1	use crate::authority::ConstitutionalAuthority;
     2	use amun_networking::validator_certificate::ValidatorCertificate;
     3	use serde::{Deserialize, Serialize};
     4	use std::collections::BTreeMap;
     5	
     6	#[derive(Debug, Clone, Serialize, Deserialize, Default)]
     7	pub struct AuthorityRegistry {
     8	    authorities: BTreeMap<u64, ConstitutionalAuthority>,
     9	    active_version: u64,
    10	    pub transition: Option<AuthorityTransition>,
    11	}
    12	
    13	/// Describes a scheduled transition from one authority version to another.
    14	#[derive(Debug, Clone, Serialize, Deserialize)]
    15	pub struct AuthorityTransition {
    16	    pub from_version: u64,
    17	    pub to_version: u64,
    18	    pub activation_height: u64,
    19	    pub grace_period_blocks: u64,
    20	}
    21	
    22	impl AuthorityRegistry {
    23	    pub fn new() -> Self {
    24	        Self::default()
    25	    }
    26	
    27	    pub fn active(&self) -> Option<&ConstitutionalAuthority> {
    28	        self.authorities.get(&self.active_version)
    29	    }
    30	
    31	    pub fn by_version(&self, version: u64) -> Option<&ConstitutionalAuthority> {
    32	        self.authorities.get(&version)
    33	    }
    34	
    35	    pub fn register(&mut self, authority: ConstitutionalAuthority) {
    36	        let version = authority.authority_version;
    37	        self.authorities.insert(version, authority);
    38	        if version > self.active_version {
    39	            self.active_version = version;
    40	        }
    41	    }
    42	
    43	    pub fn revoke(&mut self, version: u64) {
    44	        if let Some(auth) = self.authorities.get_mut(&version) {
    45	            auth.revoked = true;
    46	        }
    47	    }
    48	
    49	    /// Create a registry pre-populated with a single genesis authority.
    50	    pub fn from_genesis(authority: ConstitutionalAuthority) -> Self {
    51	        let mut registry = Self::new();
    52	        registry.register(authority);
    53	        registry
    54	    }
    55	
    56	    pub fn is_revoked(&self, version: u64) -> bool {
    57	        self.authorities.get(&version).is_some_and(|a| a.revoked)
    58	    }
    59	
    60	    /// Activate a new authority, making it the active one.
    61	    pub fn activate(&mut self, authority: ConstitutionalAuthority) {
    62	        let version = authority.authority_version;
    63	        self.authorities.insert(version, authority);
    64	        self.active_version = version;
    65	    }
    66	
    67	    /// Retire an authority so it can no longer issue new certificates.
    68	    pub fn retire(&mut self, authority_version: u64, height: u64) {
    69	        if let Some(auth) = self.authorities.get_mut(&authority_version) {
    70	            auth.revoked = true;
    71	            auth.revoked_at_height = Some(height);
    72	        }
    73	    }
    74	
    75	    /// Schedule a transition to a new authority version.
    76	    pub fn schedule_transition(&mut self, transition: AuthorityTransition) {
    77	        self.transition = Some(transition);
    78	    }
    79	
    80	    /// Return the authority (or authorities) valid at a given block height.
    81	    pub fn valid_authorities_at(&self, height: u64) -> Vec<&ConstitutionalAuthority> {
    82	        let mut result = Vec::new();
    83	        if let Some(ref t) = self.transition {
    84	            if height < t.activation_height {
    85	                // Pre-activation: only the old authority is valid
    86	                if let Some(a) = self.by_version(t.from_version) {
    87	                    result.push(a);
    88	                }
    89	            } else if height < t.activation_height + t.grace_period_blocks {
    90	                // Grace period: both authorities are valid
    91	                if let Some(a) = self.by_version(t.from_version) {
    92	                    result.push(a);
    93	                }
    94	                if let Some(a) = self.by_version(t.to_version) {
    95	                    result.push(a);
    96	                }
    97	            } else {
    98	                // Post-grace: only the new authority is valid
    99	                if let Some(a) = self.by_version(t.to_version) {
   100	                    result.push(a);
   101	                }
   102	            }
   103	        } else {
   104	            // No transition scheduled: return the active authority
   105	            if let Some(a) = self.active() {
   106	                result.push(a);
   107	            }
   108	        }
   109	        result
   110	    }
   111	
   112	    /// Check if an authority can issue new certificates at a given height.
   113	    /// Verify a certificate against the authority registry at a given block height.
   114	    /// Follows the Authority Sunset Model: the authority must be currently valid
   115	    /// at the verification height, not just at issuance time.
   116	    pub fn verify_certificate_at(&self, cert: &ValidatorCertificate, height: u64) -> bool {
   117	        // 1. Authority version must exist
   118	        let authority = match self.by_version(cert.authority_version) {
   119	            Some(a) => a,
   120	            None => return false,
   121	        };
   122	
   123	        // 2. Authority ID must match (skip for legacy certs with zero ID)
   124	        if cert.authority_id != [0u8; 32] && authority.authority_id != cert.authority_id {
   125	            return false;
   126	        }
   127	
   128	        // 3. Authority must not be revoked at this height
   129	        if let Some(revoked_height) = authority.revoked_at_height {
   130	            if height >= revoked_height {
   131	                return false;
   132	            }
   133	        }
   134	
   135	        // 4. Authority must be valid at this height (transition/grace window check)
   136	        let valid = self.valid_authorities_at(height);
   137	        if !valid
   138	            .iter()
   139	            .any(|a| a.authority_version == cert.authority_version)
   140	        {
   141	            return false;
   142	        }
   143	
   144	        // 5. Cryptographic signature must verify
   145	        cert.verify(&authority.authority_public_key)
   146	    }
   147	
   148	    pub fn can_issue_at(&self, authority_version: u64, height: u64) -> bool {
   149	        if self.is_revoked(authority_version) {
   150	            return false;
   151	        }
   152	        if let Some(ref t) = self.transition {
   153	            // After activation, the old authority cannot issue new certs
   154	            if authority_version == t.from_version && height >= t.activation_height {
   155	                return false;
   156	            }
   157	        }
   158	        // The active authority can always issue (if not revoked)
   159	        if let Some(active) = self.active() {
   160	            if active.authority_version == authority_version {
   161	                return true;
   162	            }
   163	        }
   164	        false
   165	    }
   166	}

     4	/// A registry of trusted constitutional authorities.
     5	///
     6	/// Trust anchors are the root of cryptographic trust in the network.
     7	/// They issue ValidatorCertificates that bind peer identities to public keys.
     8	#[derive(Debug, Clone, Default)]
     9	pub struct TrustAnchorRegistry {
    10	    anchors: BTreeMap<PeerId, [u8; 32]>, // peer_id -> public_key
    11	}
    12	
    13	impl TrustAnchorRegistry {
    14	    pub fn new() -> Self {
    15	        Self::default()
    16	    }
    17	
    18	    /// Register a trust anchor.
    19	    pub fn register(&mut self, peer_id: PeerId, public_key: [u8; 32]) {
    20	        self.anchors.insert(peer_id, public_key);
    21	    }
    22	
    23	    /// Revoke a trust anchor.
    24	    pub fn revoke(&mut self, peer_id: &PeerId) -> bool {
    25	        self.anchors.remove(peer_id).is_some()
    26	    }
    27	
    28	    /// Check if a peer is a trusted authority.
    29	    pub fn is_trusted(&self, peer_id: &PeerId) -> bool {
    30	        self.anchors.contains_key(peer_id)
    31	    }
    32	
    33	    /// Get the public key of a trust anchor.
    34	    pub fn get_key(&self, peer_id: &PeerId) -> Option<&[u8; 32]> {
    35	        self.anchors.get(peer_id)
    36	    }
    37	
    38	    /// Check if the registry is empty.
    39	    pub fn is_empty(&self) -> bool {
    40	        self.anchors.is_empty()
    41	    }
    42	
    43	    /// Number of registered trust anchors.
    44	    pub fn len(&self) -> usize {
    45	        self.anchors.len()
    46	    }
    47	}
