     8	/// Handshake message sent when two nodes establish a connection.
     9	#[derive(Debug, Clone, Serialize, Deserialize)]
    10	pub struct HandshakeMessage {
    11	    pub peer_id: PeerId,
    12	    pub public_key: [u8; 32],
    13	    pub certificate: ValidatorCertificate,
    14	    pub genesis_hash: [u8; 32],
    15	    pub protocol_version: u32,
    16	    pub node_name: String,
    17	    pub listen_port: u16,
    18	}
    19	
    20	impl HandshakeMessage {
    21	    pub fn new(
    22	        keypair: &PeerKeyPair,
    23	        certificate: &ValidatorCertificate,
    24	        genesis_hash: [u8; 32],
    25	        node_name: &str,
    26	        listen_port: u16,
    27	    ) -> Self {
    28	        Self {
    29	            peer_id: keypair.peer_id(),
    30	            public_key: keypair.verifying_key.to_bytes(),
    31	            certificate: certificate.clone(),
    32	            genesis_hash,
    33	            protocol_version: 1,
    34	            node_name: node_name.to_string(),
    35	            listen_port,
    36	        }
    37	    }
    38	
    39	    /// Verify the handshake message against our genesis and certificate expectations.
    40	    pub fn verify(&self, our_genesis_hash: &[u8; 32]) -> Result<(), String> {
    41	        // Verify genesis hash matches
    42	        if self.genesis_hash != *our_genesis_hash {
    43	            return Err(format!(
    44	                "Genesis hash mismatch: peer has {} but we have {}",
    45	                hex::encode(self.genesis_hash),
    46	                hex::encode(*our_genesis_hash)
    47	            ));
    48	        }
    49	
    50	        // Verify certificate is self-consistent
    51	        if *self.certificate.validator_id() != self.peer_id {
    52	            return Err("Certificate validator_id does not match peer_id".into());
    53	        }
    54	
    55	        if *self.certificate.public_key() != self.public_key {
    56	            return Err("Certificate public_key does not match handshake public_key".into());
    57	        }
    58	
    59	        // Verify certificate signature
    60	        if !self.certificate.verify(&self.public_key) {
    61	            return Err("Certificate signature verification failed".into());
    62	        }
    63	
    64	        Ok(())
    65	    }
    66	}

    83	            node_name: handshake.node_name.clone(),
    84	            listen_port: handshake.listen_port,
    85	            protocol_version: handshake.protocol_version,

    87	    }
    88	}
