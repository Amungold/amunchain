    38	    /// Verify the handshake message against our genesis and certificate expectations.
    39	    pub fn verify(&self, our_genesis_hash: &[u8; 32]) -> Result<(), String> {
    40	        // Verify genesis hash matches
    41	        if self.genesis_hash != *our_genesis_hash {
    42	            return Err(format!(
    43	                "Genesis hash mismatch: peer has {} but we have {}",
    44	                hex::encode(self.genesis_hash),
    45	                hex::encode(*our_genesis_hash)
    46	            ));
    47	        }
    48	
    49	        // Verify certificate is self-consistent
    50	        if self.certificate.validator_id != self.peer_id {
    51	            return Err("Certificate validator_id does not match peer_id".into());
    52	        }
    53	
    54	        if self.certificate.public_key != self.public_key {
    55	            return Err("Certificate public_key does not match handshake public_key".into());
    56	        }
    57	
    58	        // Verify certificate signature
    59	        if !self.certificate.verify(&self.public_key) {
    60	            return Err("Certificate signature verification failed".into());
    61	        }
    62	
    63	        Ok(())
    64	    }
