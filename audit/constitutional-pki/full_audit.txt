     5	//! Constitutional authority layer.
     6	//!
     7	//! This crate provides the primitives for building replay-safe,
     8	//! self-certifying authority chains on top of the constitutional
     9	//! signing infrastructure.  Every certificate is content-addressed
    10	//! via a domain-separated BLAKE3 digest computed over a stable
    11	//! subset of its fields, avoiding circular self-reference.
    12	//!
    13	//! # Organisation
    14	//! - `certificate`  – the core `ConstitutionalCertificate` type
    15	//! - `chain`        – cryptographically verified certificate chains
    16	//! - `revocation`   – deterministic, append-only revocation registry
    17	//! - `rotation`     – key-rotation semantics with signed proofs
    18	//! - `trust`        – frozen trust anchors

     8	/// A constitutional certificate whose identity is derived from a
     9	/// stable subset of its fields (everything except `certificate_id`,
    10	/// `issuer`, and `subject`).  This guarantees that the certificate
    11	/// can reference its own identity without creating a circular
    12	/// dependency.
    13	#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
    14	#[serde(rename_all = "snake_case")]
    15	pub struct ConstitutionalCertificate {
    16	    pub schema_version: u32,
    17	    pub certificate_id: String, // hex BLAKE3 of identity bytes
    18	    pub issuer: String,         // certificate_id of the issuer
    19	    pub subject: String,        // certificate_id of the subject
    20	    pub subject_verifying_key_hex: String,
    21	    pub lineage_parent_hash: Option<String>,
    22	    pub epoch_start: String,
    23	    pub epoch_end: String,
    24	    pub constitutional_scope: String,
    25	    pub timestamp: String,
    26	}
    27	
    28	impl ArtifactDigest for ConstitutionalCertificate {
    29	    fn domain_separator(&self) -> &'static [u8] {
    30	        b"AMUN_CERTIFICATE_V1"
    31	    }
    32	
    33	    fn constitutional_digest(&self) -> [u8; 32] {
    34	        let mut hasher = blake3::Hasher::new();
    35	        hasher.update(self.domain_separator());
    36	        hasher.update(&self.identity_bytes());
    37	        *hasher.finalize().as_bytes()
    38	    }
    39	}
    40	
    41	impl ConstitutionalCertificate {
    42	    // ---------------------------------------------------------------
    43	    // Constructors – roles are separated so invariants are clear
    44	    // ---------------------------------------------------------------
    45	
    46	    /// Create a self-signed root certificate.
    47	    pub fn new_root(
    48	        subject_public_key_hex: String,
    49	        epoch_start: String,
    50	        epoch_end: String,
    51	        scope: String,
    52	        timestamp: String,
    53	    ) -> Self {
    54	        let mut tmp = Self::empty();
    55	        tmp.subject_verifying_key_hex = subject_public_key_hex;
    56	        tmp.epoch_start = epoch_start;
    57	        tmp.epoch_end = epoch_end;
    58	        tmp.constitutional_scope = scope;
    59	        tmp.timestamp = timestamp;
    60	
    61	        let id = tmp.compute_id();
    62	        tmp.certificate_id = id.clone();
    63	        tmp.issuer = id.clone();
    64	        tmp.subject = id;
    65	        tmp
    66	    }
    67	
    68	    /// Create a child certificate issued by a parent.
    69	    pub fn new_child(
    70	        issuer_id: String, // parent certificate_id
    71	        subject_public_key_hex: String,
    72	        lineage_parent: String, // same as issuer_id
    73	        epoch_start: String,
    74	        epoch_end: String,
    75	        scope: String,
    76	        timestamp: String,
    77	    ) -> Self {
    78	        let mut tmp = Self::empty();
    79	        tmp.issuer = issuer_id;
    80	        tmp.subject_verifying_key_hex = subject_public_key_hex;
    81	        tmp.lineage_parent_hash = Some(lineage_parent);
    82	        tmp.epoch_start = epoch_start;
    83	        tmp.epoch_end = epoch_end;
    84	        tmp.constitutional_scope = scope;
    85	        tmp.timestamp = timestamp;
    86	
    87	        let id = tmp.compute_id();
    88	        tmp.certificate_id = id.clone();
    89	        tmp.subject = id;
    90	        tmp
    91	    }
    92	
    93	    // ---------------------------------------------------------------
    94	    // Identity helpers
    95	    // ---------------------------------------------------------------
    96	
    97	    /// Build a bare certificate used as a template for identity computation.
    98	    fn empty() -> Self {
    99	        Self {
   100	            schema_version: 1,
   101	            certificate_id: String::new(),
   102	            issuer: String::new(),
   103	            subject: String::new(),
   104	            subject_verifying_key_hex: String::new(),
   105	            lineage_parent_hash: None,
   106	            epoch_start: String::new(),
   107	            epoch_end: String::new(),
   108	            constitutional_scope: String::new(),
   109	            timestamp: String::new(),
   110	        }
   111	    }
   112	
   113	    /// Compute the certificate id from the canonical identity bytes.
   114	    fn compute_id(&self) -> String {
   115	        let bytes = self.identity_bytes();
   116	        let mut hasher = blake3::Hasher::new();
   117	        hasher.update(b"AMUN_CERTIFICATE_V1");
   118	        hasher.update(&bytes);
   119	        hex::encode(hasher.finalize().as_bytes())
   120	    }
   121	
   122	    /// Canonical bytes used for identity hashing.
   123	    ///
   124	    /// This deliberately omits `certificate_id`, `issuer`, and `subject`
   125	    /// because those fields depend on the identity itself for root
   126	    /// certificates, which would create a circular dependency.
   127	    fn identity_bytes(&self) -> Vec<u8> {
   128	        let mut c = self.clone();
   129	        c.certificate_id = String::new();
   130	        c.issuer = String::new();
   131	        c.subject = String::new();
   132	        serde_json::to_vec(&c).expect("Identity serialization must not fail")
   133	    }
   134	
   135	    /// Convenience: return the full artifact digest (including all fields).
   136	    pub fn full_digest_hex(&self) -> String {
   137	        self.digest_hex()
   138	    }
   139	}
   140	
   141	impl CanonicalSerialize for ConstitutionalCertificate {
   142	    fn canonical_bytes(&self) -> Vec<u8> {
   143	        serde_json::to_vec(self).expect("Canonical serialization must not fail")
   144	    }
   145	}

     9	/// A cryptographically verified chain of signed constitutional certificates.
    10	pub struct CertificateChain {
    11	    pub certificates: Vec<SignedArtifact<ConstitutionalCertificate>>,
    12	}
    13	
    14	impl CertificateChain {
    15	    pub fn new(root: SignedArtifact<ConstitutionalCertificate>) -> Self {
    16	        Self {
    17	            certificates: vec![root],
    18	        }
    19	    }
    20	
    21	    /// Append a signed certificate that chains to the current tail.
    22	    pub fn append(
    23	        &mut self,
    24	        cert: SignedArtifact<ConstitutionalCertificate>,
    25	    ) -> Result<(), String> {
    26	        let tail = &self.certificates.last().ok_or("Empty chain")?.artifact;
    27	        if cert.artifact.lineage_parent_hash != Some(tail.certificate_id.clone()) {
    28	            return Err("Lineage parent mismatch".into());
    29	        }
    30	        self.certificates.push(cert);
    31	        Ok(())
    32	    }
    33	
    34	    /// Look up the public verifying key for a given certificate id.
    35	    fn resolve_key(&self, cert_id: &str) -> Option<String> {
    36	        self.certificates
    37	            .iter()
    38	            .find(|s| s.artifact.certificate_id == cert_id)
    39	            .map(|s| s.signature.verifying_key_hex.clone())
    40	    }
    41	
    42	    /// Full validation:
    43	    /// - every certificate is signed by its issuer's key
    44	    /// - the root is self-signed
    45	    /// - no certificate is revoked
    46	    /// - child epochs are contained within the parent's epoch
    47	    pub fn validate(&self, revocation: &RevocationRegistry) -> Result<(), String> {
    48	        for i in 0..self.certificates.len() {
    49	            let signed = &self.certificates[i];
    50	            let cert = &signed.artifact;
    51	
    52	            // 1. Revocation
    53	            if revocation.is_revoked(&cert.certificate_id) {
    54	                return Err(format!("Certificate {} revoked", cert.certificate_id));
    55	            }
    56	
    57	            // 2. Signature & authority
    58	            if i == 0 {
    59	                if cert.issuer != cert.subject {
    60	                    return Err("Root certificate is not self-signed".into());
    61	                }
    62	                signed
    63	                    .verify()
    64	                    .map_err(|e| format!("Root signature invalid: {}", e))?;
    65	            } else {
    66	                let issuer_key = self.resolve_key(&cert.issuer).ok_or_else(|| {
    67	                    format!("Issuer certificate {} not found in chain", cert.issuer)
    68	                })?;
    69	                if signed.signature.verifying_key_hex != issuer_key {
    70	                    return Err(format!(
    71	                        "Certificate {} not signed by issuer {}",
    72	                        cert.certificate_id, cert.issuer
    73	                    ));
    74	                }
    75	                signed
    76	                    .verify()
    77	                    .map_err(|e| format!("Signature invalid on {}: {}", cert.certificate_id, e))?;
    78	            }
    79	
    80	            // 3. Epoch containment
    81	            if i > 0 {
    82	                let parent = &self.certificates[i - 1].artifact;
    83	                if cert.epoch_start < parent.epoch_start || cert.epoch_end > parent.epoch_end {
    84	                    return Err(format!(
    85	                        "Epoch of {} exceeds parent epoch of {}",
    86	                        cert.certificate_id, parent.certificate_id
    87	                    ));
    88	                }
    89	            }
    90	        }
    91	        Ok(())
    92	    }
    93	}

     8	/// Enforces cryptographically verified key rotation.
     9	pub struct KeyRotationLaw;
    10	
    11	impl KeyRotationLaw {
    12	    /// Validate that `new_signed_cert` is a valid rotation from the
    13	    /// authority identified by `old_public_key_hex`.
    14	    ///
    15	    /// The new certificate must be signed by the old authority key
    16	    /// and its subject must differ from the old key.
    17	    pub fn validate_rotation(
    18	        old_public_key_hex: &str,
    19	        new_signed_cert: &SignedArtifact<ConstitutionalCertificate>,
    20	        old_keypair: &ConstitutionalKeyPair,
    21	    ) -> Result<(), String> {
    22	        new_signed_cert
    23	            .verify()
    24	            .map_err(|e| format!("Rotation signature invalid: {}", e))?;
    25	
    26	        if new_signed_cert.artifact.subject_verifying_key_hex == old_public_key_hex {
    27	            return Err("Rotated key must differ from previous key".into());
    28	        }
    29	
    30	        if old_keypair.verifying_key_hex() != old_public_key_hex {
    31	            return Err("Provided old key does not match expected public key".into());
    32	        }
    33	
    34	        Ok(())
    35	    }
    36	}

     1	use crate::keys::ConstitutionalKeyPair;
     2	use crate::signature::ConstitutionalSignature;
     3	use amun_constitution_builder::digest::ArtifactDigest;
     4	use ed25519_dalek::{Verifier, VerifyingKey};
     5	use serde::{Deserialize, Serialize};
     6	
     7	#[derive(Clone, Serialize, Deserialize)]
     8	pub struct SignedArtifact<T: ArtifactDigest> {
     9	    pub artifact: T,
    10	    pub signature: ConstitutionalSignature,
    11	}
    12	
    13	impl<T: ArtifactDigest> SignedArtifact<T> {
    14	    pub fn sign(artifact: T, keypair: &ConstitutionalKeyPair) -> Self {
    15	        let digest = artifact.constitutional_digest();
    16	        let sig = keypair.sign(&digest);
    17	        let verifying_hex = keypair.verifying_key_hex();
    18	        Self {
    19	            artifact,
    20	            signature: ConstitutionalSignature::new(sig, verifying_hex),
    21	        }
    22	    }
    23	
    24	    pub fn verify(&self) -> Result<(), String> {
    25	        let key_bytes = hex::decode(&self.signature.verifying_key_hex)
    26	            .map_err(|e| format!("Invalid verifying key hex: {}", e))?;
    27	        let arr: [u8; 32] = key_bytes
    28	            .try_into()
    29	            .map_err(|_| "Verifying key must be 32 bytes".to_string())?;
    30	        let verifying_key =
    31	            VerifyingKey::from_bytes(&arr).map_err(|e| format!("Invalid verifying key: {}", e))?;
    32	
    33	        let digest = self.artifact.constitutional_digest();
    34	        verifying_key
    35	            .verify(&digest, self.signature.dalek_signature())
    36	            .map_err(|e| format!("Signature verification failed: {}", e))
    37	    }
    38	}
    39	
    40	// Manual PartialEq + Debug to satisfy DelegationChain derives
    41	impl<T: ArtifactDigest + PartialEq> PartialEq for SignedArtifact<T> {
    42	    fn eq(&self, other: &Self) -> bool {
    43	        self.artifact == other.artifact
    44	            && self.signature.verifying_key_hex == other.signature.verifying_key_hex
    45	            && self.signature.to_hex() == other.signature.to_hex()
    46	    }
    47	}
    48	
    49	impl<T: ArtifactDigest + std::fmt::Debug> std::fmt::Debug for SignedArtifact<T> {
    50	    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    51	        f.debug_struct("SignedArtifact")
    52	            .field("artifact", &self.artifact)
    53	            .field("signature", &self.signature.verifying_key_hex)
    54	            .finish()
    55	    }
    56	}

    16	    /// When this certificate becomes valid.
    17	    pub authority_id: [u8; 32],
    18	    pub authority_version: u64,
    19	    pub valid_from: u64,
    20	    /// When this certificate expires (0 = never).
    21	    pub valid_until: u64,
    22	    /// The authority's signature over the certificate fields.
    23	    pub authority_signature: Vec<u8>,
    24	}
    25	
    26	impl ValidatorCertificate {
    27	    /// Create a new certificate signed by an authority.
    28	    pub fn issue(
    29	        validator_id: PeerId,
    30	        public_key: [u8; 32],
    31	        issuer_keypair: &PeerKeyPair,
    32	        valid_from: u64,
    33	        valid_until: u64,
    34	    ) -> Self {
    35	        let mut cert = Self {
    36	            validator_id,
    37	            public_key,
    38	            authority_id: [0u8; 32],
    39	            authority_version: 0,
    40	            issuer: issuer_keypair.peer_id(),
    41	            valid_from,
    42	            valid_until,
    43	            authority_signature: Vec::new(),
    44	        };
    45	        // Sign the certificate fields (excluding signature itself)
    46	        let payload = cert.serialize_for_signing();
    47	        cert.authority_signature = issuer_keypair.sign(&payload);
    48	        cert
    49	    }
    50	
    51	    /// Serialize certificate fields for signing (excludes the signature).
    52	    fn serialize_for_signing(&self) -> Vec<u8> {
    53	        let mut data = Vec::new();
    54	        data.extend_from_slice(&self.validator_id.0);
    55	        data.extend_from_slice(&self.public_key);
    56	        data.extend_from_slice(&self.authority_version.to_le_bytes());
    57	        data.extend_from_slice(&self.authority_id);
    58	        data.extend_from_slice(&self.issuer.0);
    59	        data.extend_from_slice(&self.valid_from.to_le_bytes());
    60	        data.extend_from_slice(&self.valid_until.to_le_bytes());
    61	        data
    62	    }
    63	
    64	    /// Verify the authority's signature on this certificate.
    65	    pub fn verify(&self, authority_public_key: &[u8; 32]) -> bool {
    66	        let payload = self.serialize_for_signing();
    67	        PeerKeyPair::verify(authority_public_key, &payload, &self.authority_signature)
    68	    }
    69	
    70	    /// Check if the certificate is currently valid.
    71	    pub fn is_valid_at(&self, timestamp: u64) -> bool {
    72	        timestamp >= self.valid_from && (self.valid_until == 0 || timestamp <= self.valid_until)
    73	    }
    74	
    75	    /// Create a new certificate with explicit authority version and id.
    76	    pub fn issue_v2(
    77	        validator_id: PeerId,
    78	        public_key: [u8; 32],
    79	        authority_version: u64,
    80	        authority_id: [u8; 32],
    81	        issuer_keypair: &PeerKeyPair,
    82	        valid_from: u64,
    83	        valid_until: u64,
    84	    ) -> Self {
    85	        let mut cert = Self {
    86	            validator_id,
    87	            public_key,
    88	            authority_version,
    89	            authority_id,
    90	            issuer: issuer_keypair.peer_id(),
    91	            valid_from,
    92	            valid_until,
    93	            authority_signature: Vec::new(),
    94	        };
    95	        let payload = cert.serialize_for_signing();
    96	        cert.authority_signature = issuer_keypair.sign(&payload);
    97	        cert
    98	    }
    99	}

    20	    pub fn new(
    21	        keypair: &PeerKeyPair,
    22	        certificate: &ValidatorCertificate,
    23	        genesis_hash: [u8; 32],
    24	        node_name: &str,
    25	        listen_port: u16,
    26	    ) -> Self {
    27	        Self {
    28	            peer_id: keypair.peer_id(),
    29	            public_key: keypair.verifying_key.to_bytes(),
    30	            certificate: certificate.clone(),
    31	            genesis_hash,
    32	            protocol_version: 1,
    33	            node_name: node_name.to_string(),
    34	            listen_port,
    35	        }
    36	    }

    38	    /// Verify the handshake message against our genesis and certificate expectations.
    39	    pub fn verify(&self, our_genesis_hash: &[u8; 32]) -> Result<(), String> {
    40	        // Verify genesis hash matches
    41	        if self.genesis_hash != *our_genesis_hash {
    42	            return Err(format!(
    43	                "Genesis hash mismatch: peer has {} but we have {}",
    44	                hex::encode(self.genesis_hash),
    45	                hex::encode(*our_genesis_hash)
    46	            ));
    47	        }
    48	
    49	        // Verify certificate is self-consistent
    50	        if self.certificate.validator_id != self.peer_id {
    51	            return Err("Certificate validator_id does not match peer_id".into());
    52	        }
    53	
    54	        if self.certificate.public_key != self.public_key {
    55	            return Err("Certificate public_key does not match handshake public_key".into());
    56	        }
    57	
    58	        // Verify certificate signature
    59	        if !self.certificate.verify(&self.public_key) {
    60	            return Err("Certificate signature verification failed".into());
    61	        }
    62	
    63	        Ok(())
    64	    }

     8	
     9	use crate::config::Config;
    10	use crate::error::{io_err, NodeError};
    11	use crate::genesis::Genesis;
    12	
    13	#[allow(dead_code)]
    14	pub struct BootstrapContext {
    15	    pub transport: TcpTransport,
    16	    pub node: NetworkNode,
    17	    pub cert: ValidatorCertificate,
    18	    pub genesis: Genesis,
    19	    pub genesis_hash: [u8; 32],
    20	    pub peer_id_bytes: [u8; 32],
    21	    pub config: Config,
    22	    pub config_dir: PathBuf,
    23	}
    24	
    25	pub fn initialize(config_path: &str) -> Result<BootstrapContext, NodeError> {
    26	    let config = Config::load(config_path)?;
    27	    let config_dir = Path::new(config_path).parent().unwrap_or(Path::new("."));
    28	    tracing::info!(node = %config.node.name, "Node configured");
    29	
    30	    // Identity
    31	    let key_path = config_dir.join(&config.identity.key_file);
    32	    let keypair = crate::identity::load_or_create_keypair(key_path.to_str().unwrap())?;
    33	    let peer_id = keypair.peer_id();
    34	    tracing::info!(peer_id = %hex::encode(peer_id.0), "Identity loaded");
    35	
    36	    // Genesis
    37	    let genesis_path = config_dir.join(&config.genesis.file);
    38	    let genesis_str = fs::read_to_string(&genesis_path).map_err(|e| io_err(&genesis_path, e))?;
    39	    let genesis: Genesis = serde_json::from_str(&genesis_str).map_err(|e| NodeError::Json {
    40	        path: genesis_path.clone(),
    41	        source: e,
    42	    })?;
    43	
    44	    if !amun_orchestrator_genesis::validator::validate(&genesis) {
    45	        return Err(NodeError::Genesis("Genesis validation failed".into()));
    46	    }
    47	
    48	    let genesis_hash = {
    49	        let mut gen = genesis.clone();
    50	        amun_orchestrator_genesis::generator::recompute_hash(&mut gen);
    51	        hex::decode(&gen.genesis_hash)
    52	            .map_err(|e| NodeError::Genesis(format!("Invalid genesis hash: {}", e)))?
    53	            .try_into()
    54	            .map_err(|_| NodeError::Genesis("Genesis hash must be 32 bytes".into()))?
    55	    };
    56	
    57	    tracing::info!(
    58	        path = %genesis_path.display(),
    59	        hash = %hex::encode(genesis_hash),
    60	        "Genesis loaded and validated"
    61	    );
    62	
    63	    // Certificate
    64	    let cert_path = config_dir.join("validator.crt");
    65	    let cert = crate::certificate_loader::load_validator_certificate(
    66	        cert_path.to_str().unwrap(),
    67	        &genesis,
    68	    )
    69	    .map_err(NodeError::Certificate)?;
    70	    tracing::info!("Validator certificate verified");
    71	
    72	    // Extract SigningKey from the loaded keypair
    73	    // Use the existing seed() method and SigningKey::from_bytes()
    74	    let seed = keypair.to_seed();
    75	    let signing_key = ed25519_dalek::SigningKey::from_bytes(&seed);
    76	
    77	    // Network ID: SHA-256 of chain_id
    78	    let network_id = {
    79	        use sha2::{Digest, Sha256};
    80	        let mut hasher = Sha256::new();
    81	        hasher.update(genesis.chain_id.as_bytes());
    82	        hasher.finalize().into()
    83	    };
    84	
    85	    // Constitution from genesis
    86	    let constitution = ConstitutionInfo {
    87	        version: 1,
    88	        hash: genesis_hash,
    89	        proof_system_version: 1,
    90	        state_commitment_algorithm: "MerklePatricia".to_string(),
    91	        accepted_features: vec!["sync".into(), "vote".into(), "block_range".into()],
    92	    };
    93	
    94	    // REAL listen address from config
    95	    let addr: std::net::SocketAddr =
    96	        format!("{}:{}", config.node.listen_host, config.node.listen_port)
    97	            .parse()
    98	            .map_err(|_| {
    99	                NodeError::InvalidAddress(format!(
   100	                    "{}:{}",
   101	                    config.node.listen_host, config.node.listen_port
   102	                ))
   103	            })?;
   104	
   105	    // NodeNetwork with REAL identity
   106	    let network = crate::network::NodeNetwork::new(
   107	        addr,
   108	        signing_key,
   109	        peer_id.0,
   110	        network_id,
   111	        genesis_hash,
   112	        constitution,
   113	    )
   114	    .map_err(NodeError::InvalidAddress)?;
   115	
   116	    let mut node = NetworkNode::new(peer_id.0);
   117	    node.keypair = Some(keypair);
   118	
   119	    Ok(BootstrapContext {
   120	        transport: network.transport,
   121	        node,
   122	        cert,
   123	        genesis,
   124	        genesis_hash,
   125	        peer_id_bytes: peer_id.0,
   126	        config,
   127	        config_dir: config_dir.to_path_buf(),
   128	    })
   129	}
